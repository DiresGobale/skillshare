   230→// ============================================================
   231→// MOBILE NAVBAR — slide-in panel controlled by hamburger toggle
   232→// ============================================================
   233→function MobileNavbar({
   234→  open,
   235→  onClose,
   236→  role,
   237→  view,
   238→  onNavigate,
   239→}: {
   240→  open: boolean;
   241→  onClose: () => void;
   242→  role: Role;
   243→  view: View;
   244→  onNavigate: (v: View) => void;
   245→}) {
   246→  const nav = getNav(role);
   247→  const { user } = useApp();
   248→  const { data: profileData } = useFetch<any>("/api/profile");
   249→  const profilePhoto = profileData?.profile?.profilePhoto;
   250→  useEffect(() => {
   251→    if (open) {
   252→      document.body.style.overflow = "hidden";
   253→    } else {
   254→      document.body.style.overflow = "";
   255→    }
   256→    return () => {
   257→      document.body.style.overflow = "";
   258→    };
   259→  }, [open]);
   260→
   261→  // Close on Escape
   262→  useEffect(() => {
   263→    if (!open) return;
   264→    const handler = (e: KeyboardEvent) => {
   265→      if (e.key === "Escape") onClose();
   266→    };
   267→    window.addEventListener("keydown", handler);
   268→    return () => window.removeEventListener("keydown", handler);
   269→  }, [open, onClose]);
   270→
   271→  return (
   272→    <>
   273→      {/* Backdrop */}
   274→      <div
   275→        className={cn(
   276→          "fixed inset-0 z-40 bg-black/50 backdrop-blur-sm transition-opacity duration-300 lg:hidden",
   277→          open ? "opacity-100" : "pointer-events-none opacity-0"
   278→        )}
   279→        onClick={onClose}
   280→        aria-hidden="true"
   281→      />
   282→
   283→      {/* Slide-in panel */}
   284→      <nav
   285→        id="mobile-navbar"
   286→        aria-label="Mobile navigation"
   287→        className={cn(
   288→          "fixed inset-y-0 left-0 z-50 flex w-72 max-w-[85vw] flex-col border-r border-sidebar-border bg-sidebar shadow-xl transition-transform duration-300 ease-in-out lg:hidden",
   289→          open ? "translate-x-0" : "-translate-x-full"
   290→        )}
   291→      >
   292→        {/* Header with brand + close button */}
   293→        <div className="flex h-16 items-center justify-between border-b border-sidebar-border px-5">
   294→          <BrandMark />
   295→          <button
   296→            onClick={onClose}
   297→            aria-label="Close menu"
   298→            className="inline-flex h-8 w-8 items-center justify-center rounded-md text-muted-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   299→          >
   300→            <X className="h-4 w-4" />
   301→          </button>
   302→        </div>
   303→
   304→        {/* Nav items */}
   305→        <div className="flex-1 overflow-y-auto px-3 py-4">
   306→          <p className="px-3 pb-2 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   307→            {role === "seeker" ? "Job seeker" : role === "employer" ? "Employer" : "Administrator"}
   308→          </p>
   309→          <div className="space-y-0.5">
   310→            {nav.map((item) => {
   311→              const active = view === item.view;
   312→              return (
   313→                <button
   314→                  key={item.view}
   315→                  onClick={() => {
   316→                    onNavigate(item.view);
   317→                    onClose();
   318→                  }}
   319→                  className={cn(
   320→                    "group flex w-full items-center gap-2.5 rounded-md px-3 py-2.5 text-sm font-medium transition-colors",
   321→                    active
   322→                      ? "bg-primary text-primary-foreground shadow-sm"
   323→                      : "text-sidebar-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   324→                  )}
   325→                >
   326→                  <span className={cn(active ? "text-primary-foreground" : "text-muted-foreground group-hover:text-sidebar-accent-foreground")}>
   327→                    {item.icon}
   328→                  </span>
   329→                  <span className="flex-1 text-left">{item.label}</span>
   330→                  {item.badge && (
   331→                    <span
   332→                      className={cn(
   333→                        "inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[10px] font-semibold",
   334→                        active ? "bg-primary-foreground/20 text-primary-foreground" : "bg-primary text-primary-foreground"
   335→                      )}
   336→                    >
   337→                      {item.badge}
   338→                    </span>
   339→                  )}
   340→                </button>
   341→              );
   342→            })}
   343→          </div>
   344→        </div>
   345→
   346→        {/* User card */}
   347→        {user && (
   348→          <div className="border-t border-sidebar-border p-3">
   349→            <div className="flex items-center gap-2.5 rounded-md px-2 py-2">
   350→              <Avatar initials={user.initials ?? user.name.slice(0,2).toUpperCase()} color={user.avatarColor} size="sm" photo={profilePhoto} />
   351→              <div className="min-w-0 flex-1">
   352→                <p className="truncate text-xs font-semibold">{user.name}</p>
   353→                <p className="truncate text-[10px] text-muted-foreground">{user.title ?? user.email}</p>
   354→              </div>
   355→            </div>
   356→          </div>
   357→        )}
   358→      </nav>
   359→    </>
   360→  );
   361→}

   504→          </div>
   505→        )}
   506→        <Button
   507→          variant="ghost"
   508→          size="icon"
   509→          className="h-9 w-9 text-muted-foreground hover:text-foreground"
   510→          onClick={() => signOut()}
   511→          aria-label="Sign out"
   512→        >
   513→          <LogOut className="h-4 w-4" />
   514→        </Button>
   515→      </div>
   516→    </header>
   517→  );
   518→}

   580→// ============================================================
   581→// MAIN LAYOUT — sidebar + topbar + content
   582→// ============================================================
   583→export function AppShell({ children }: { children: React.ReactNode }) {
   584→  const { role, view, go } = useApp();
   585→  const [mobileOpen, setMobileOpen] = useState(false);
   586→  const [showNavHint, setShowNavHint] = useState(false);
   587→
   588→  // After login, briefly auto-open the mobile sidebar so users on small
   589→  // screens can see that navigation exists. We check a sessionStorage flag
   590→  // set by the login flow, then clear it. Only triggers on mobile (lg down).
   591→  useEffect(() => {
   592→    try {
   593→      if (sessionStorage.getItem("skillmatch.justLoggedIn") === "1") {
   594→        sessionStorage.removeItem("skillmatch.justLoggedIn");
   595→        // Only auto-open on mobile (window width < 1024 = lg breakpoint)
   596→        if (typeof window !== "undefined" && window.innerWidth < 1024) {
   597→          // Small delay so the dashboard content renders first
   598→          const t1 = setTimeout(() => setMobileOpen(true), 600);
   599→          // Auto-close after 4 seconds if the user hasn't interacted
   600→          const t2 = setTimeout(() => setMobileOpen(false), 4600);
   601→          // Show a pulsing hint on the hamburger after the sidebar closes
   602→          const t3 = setTimeout(() => setShowNavHint(true), 4800);
   603→          const t4 = setTimeout(() => setShowNavHint(false), 10000);
   604→          return () => {
   605→            clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4);
   606→          };
   607→        }
   608→      }
   609→    } catch { /* ignore */ }
   610→  }, []);
   611→
   612→  return (
   613→    <div className="flex h-screen overflow-hidden bg-background">
   614→      {/* Desktop sidebar — outside the scroll container, so it stays fixed naturally */}
   615→      <aside className="hidden w-64 shrink-0 border-r border-sidebar-border bg-sidebar lg:flex lg:flex-col">
   616→        <SidebarNav role={role} view={view} onNavigate={(v) => go(v)} />
   617→      </aside>
   618→
   619→      {/* Mobile navbar (toggle-controlled, slide-in) */}
   620→      <MobileNavbar
   621→        open={mobileOpen}
   622→        onClose={() => setMobileOpen(false)}
   623→        role={role}
   624→        view={view}
   625→        onNavigate={(v) => go(v)}
   626→      />
   627→
   628→      {/* Main column — the only element that scrolls */}
   629→      <div className="flex min-w-0 flex-1 flex-col overflow-hidden">
   630→        {/* Topbar — outside the inner scroll container, so it stays fixed naturally */}
   631→        <Topbar
   632→          onToggleMobile={() => { setMobileOpen((v) => !v); setShowNavHint(false); }}
   633→          mobileOpen={mobileOpen}
   634→          showNavHint={showNavHint}
   635→        />
   636→        {/* Only this <main> scrolls; the sidebar (lg) and topbar (all sizes) stay put.
   637→            pb-20 on mobile to clear the bottom nav bar; lg:pb-0 to remove it on desktop. */}
   638→        <main id="app-main-scroll" className="flex-1 overflow-y-auto overflow-x-hidden pb-20 lg:pb-0">{children}</main>
   639→      </div>
   640→
   641→      {/* Mobile bottom navigation — fixed, persistent quick access to key pages */}
   642→      <MobileBottomNav role={role} view={view} onNavigate={(v) => go(v)} />
   643→    </div>
   644→  );
   645→}
