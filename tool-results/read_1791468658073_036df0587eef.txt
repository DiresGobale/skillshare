   108→function timeAgo(date: string) {
   109→  const d = new Date(date);
   110→  const days = Math.floor((Date.now() - d.getTime()) / (1000 * 60 * 60 * 24));
   111→  if (days === 0) return "today";
   112→  if (days === 1) return "1d ago";
   113→  if (days < 7) return `${days}d ago`;
   114→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   115→  return `${Math.floor(days / 30)}mo ago`;
   116→}

   269→export function EmployerCompanyProfile() {
   270→  const { user: sessionUser } = useApp();
   271→  const { data, loading, refetch } = useFetch<any>("/api/profile");
   272→  const profile = data?.profile;
   273→
   274→  const [name, setName] = useState("");
   275→  const [company, setCompany] = useState("");
   276→  const [title, setTitle] = useState("");
   277→  const [location, setLocation] = useState("");
   278→  const [about, setAbout] = useState("");
   279→  const [saving, setSaving] = useState(false);
   280→  const [saveMsg, setSaveMsg] = useState<{ ok: boolean; msg: string } | null>(null);
   281→
   282→  useEffect(() => {
   283→    if (profile) {
   284→      setName(profile.name ?? "");
   285→      setCompany(profile.company ?? "");
   286→      setTitle(profile.title ?? "");
   287→      setLocation(profile.location ?? "");
   288→      setAbout(profile.about ?? "");
   289→    }
   290→  }, [profile]);
   291→
   292→  const save = async () => {
   293→    setSaving(true);
   294→    setSaveMsg(null);
   295→    try {
   296→      const res = await fetch("/api/profile", {
   297→        method: "PUT",
   298→        headers: { "Content-Type": "application/json" },
   299→        body: JSON.stringify({ name, title, location, about, company }),
   300→      });
   301→      const json = await res.json();
   302→      if (json.ok) {
   303→        setSaveMsg({ ok: true, msg: "Company profile updated successfully." });
   304→        refetch();
   305→      } else {
   306→        setSaveMsg({ ok: false, msg: json.error ?? "Save failed." });
   307→      }
   308→    } catch {
   309→      setSaveMsg({ ok: false, msg: "Network error." });
   310→    } finally {
   311→      setSaving(false);
   312→    }
   313→  };
   314→
   315→  if (loading) return <PageContainer><LoadingState label="Loading company profile…" rows={3} /></PageContainer>;

   328→          />
   329→        }
   330→      />
   331→
   332→      {saveMsg && (
   333→        <div className={cn(
   334→          "mt-4 flex items-center gap-2 rounded-lg border px-4 py-3 text-sm",

   336→        )}>
   337→          {saveMsg.ok ? <CheckCircle2 className="h-4 w-4" /> : <AlertCircle className="h-4 w-4" />}
   338→          {saveMsg.msg}
   339→        </div>

   343→        {/* Company card */}
