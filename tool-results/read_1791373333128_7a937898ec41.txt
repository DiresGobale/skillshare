    82→    const handler = (e: KeyboardEvent) => {
    83→      if (e.key === "Escape") setMobileOpen(false);

    87→      document.body.style.overflow = "";

   118→              </Button>
   119→            ))}
   120→          </nav>
   121→
   122→          <div className="flex items-center gap-2">
   123→            <Button variant="ghost" size="sm" onClick={() => onAuth("login")} className="hidden sm:inline-flex">

   149→        <div className="space-y-1 p-3">

  1093→function timeAgo(date: string) {
  1094→  const d = new Date(date);
  1095→  const days = Math.floor((Date.now() - d.getTime()) / (1000 * 60 * 60 * 24));
  1096→  if (days === 0) return "today";
  1097→  if (days === 1) return "1d ago";
  1098→  if (days < 7) return `${days}d ago`;
  1099→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
  1100→  return `${Math.floor(days / 30)}mo ago`;
  1101→}
