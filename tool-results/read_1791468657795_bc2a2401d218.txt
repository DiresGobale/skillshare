   123→function timeAgo(date: string) {
   124→  const d = new Date(date);
   125→  const days = Math.floor((Date.now() - d.getTime()) / (1000 * 60 * 60 * 24));
   126→  if (days === 0) return "today";
   127→  if (days === 1) return "1d ago";
   128→  if (days < 7) return `${days}d ago`;
   129→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   130→  return `${Math.floor(days / 30)}mo ago`;
   131→}
