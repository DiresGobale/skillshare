   192→// Mock bar chart
   193→function MockBarChart({ data, color }: { data: { label: string; value: number }[]; color: string }) {
   194→  const max = Math.max(...data.map((d) => d.value));
   195→  return (
   196→    <div className="flex h-44 items-end justify-between gap-2">
   197→      {data.map((d) => (
   198→        <div key={d.label} className="flex flex-1 flex-col items-center gap-1.5">
   199→          <div className="flex w-full flex-1 items-end">
   200→            <div
   201→              className="w-full rounded-t-md transition-all hover:opacity-80"
   202→              style={{ height: `${(d.value / max) * 100}%`, background: color, minHeight: 4 }}
   203→              title={`${d.value.toLocaleString()} users`}
   204→            />
   205→          </div>
   206→          <span className="text-[10px] text-muted-foreground">{d.label}</span>
   207→        </div>
   208→      ))}
   209→    </div>
   210→  );
   211→}

   216→export function AdminUsers() {
   217→  const { toast } = useToast();
   218→  const [filter, setFilter] = useState("all");
   219→  const [roleFilter, setRoleFilter] = useState("all");
   220→  const [query, setQuery] = useState("");
   221→  const [debouncedQuery, setDebouncedQuery] = useState("");
   222→  const [selected, setSelected] = useState<any | null>(null);
   223→  const [confirmDelete, setConfirmDelete] = useState<any | null>(null);
   224→  const [confirmSuspend, setConfirmSuspend] = useState<any | null>(null);
   225→  const [page, setPage] = useState(1);
   226→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   227→  const PAGE_SIZE = 8;
   228→
   229→  // Invite user dialog state
   230→  const [inviteOpen, setInviteOpen] = useState(false);
   231→  const [inviteEmail, setInviteEmail] = useState("");
   232→  const [inviteName, setInviteName] = useState("");
   233→  const [inviteRole, setInviteRole] = useState("seeker");
   234→  const [inviting, setInviting] = useState(false);
   235→
   236→  // Build the API URL with filters
   237→  const apiUrl = `/api/admin/users?role=${roleFilter}&status=${filter}&q=${encodeURIComponent(debouncedQuery)}`;
   238→  const { data, loading, refetch } = useFetch<any>(apiUrl);
   239→
   240→  const allUsers: any[] = data?.users ?? [];
   241→
   242→  // Debounce search input for efficiency on large datasets
   243→  useEffect(() => {
   244→    const id = setTimeout(() => {
   245→      setDebouncedQuery(query);
   246→      setPage(1);
   247→    }, 250);
   248→    return () => clearTimeout(id);
   249→  }, [query]);
   250→
   251→  // Paginate — only render the current page's slice
   252→  const totalPages = Math.max(1, Math.ceil(allUsers.length / PAGE_SIZE));
   253→  const currentPage = Math.min(page, totalPages);
   254→  const paged = allUsers.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE);
   255→
   256→  const updateStatus = async (userId: string, status: string) => {
   257→    setActionLoading(userId + status);
   258→    try {
   259→      await fetch(`/api/admin/users/${userId}`, {
   260→        method: "PATCH",
   261→        headers: { "Content-Type": "application/json" },
   262→        body: JSON.stringify({ status }),
   263→      });
   264→      refetch();
   265→    } finally {
   266→      setActionLoading(null);
   267→      setConfirmSuspend(null);
   268→    }
   269→  };
   270→
   271→  const deleteUser = async (userId: string) => {
   272→    setActionLoading(userId + "delete");
   273→    try {
   274→      await fetch(`/api/admin/users/${userId}`, { method: "DELETE" });
   275→      refetch();
   276→    } finally {
   277→      setActionLoading(null);
   278→      setConfirmDelete(null);
   279→    }
   280→  };
   281→
   282→  const sendInvite = async () => {
   283→    if (!inviteEmail.trim() || !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(inviteEmail)) {
   284→      toast({ title: "Invalid email", description: "Please enter a valid email address." });
   285→      return;
   286→    }
   287→    setInviting(true);
   288→    try {
   289→      // Create an audit log entry for the invitation
   290→      await fetch("/api/admin/audit", {
   291→        method: "POST",
   292→        headers: { "Content-Type": "application/json" },
   293→        body: JSON.stringify({
   294→          action: `Invited ${inviteRole}: ${inviteEmail}`,
   295→          target: inviteEmail,
   296→          type: "admin",
   297→          status: "success",
   298→        }),
   299→      }).catch(() => { /* audit failure is non-blocking */ });
   300→
   301→      toast({
   302→        title: "Invitation sent",
   303→        description: `${inviteName || inviteEmail} has been invited as ${inviteRole}. They'll receive an email with signup instructions.`,
   304→      });
   305→      setInviteOpen(false);
   306→      setInviteEmail("");
   307→      setInviteName("");
   308→      setInviteRole("seeker");
   309→    } catch {
   310→      toast({ title: "Failed to send invite", description: "Please try again." });
   311→    } finally {
   312→      setInviting(false);
   313→    }
   314→  };
   315→
   316→  return (
   317→    <PageContainer>
   318→      <SectionHeader
   319→        title="User Management"
   320→        description="Manage job seekers, employers, and platform access."
   321→        actions={
   322→          <Dialog open={inviteOpen} onOpenChange={(o) => { setInviteOpen(o); if (!o) { setInviteEmail(""); setInviteName(""); setInviteRole("seeker"); } }}>
   323→            <DialogTrigger asChild>
   324→              <Button className="btn-rainbow"><Plus className="h-4 w-4" /> Invite user</Button>
   325→            </DialogTrigger>
   326→            <DialogContent>
   327→              <DialogHeader>
   328→                <DialogTitle>Invite a new user</DialogTitle>
   329→                <DialogDescription>
   330→                  Send an invitation to join the platform. They'll receive an email with a signup link pre-filled with their role.
   331→                </DialogDescription>
   332→              </DialogHeader>
   333→              <div className="space-y-3 py-2">
   334→                <div className="space-y-1.5">
   335→                  <Label htmlFor="invite-email">Email address *</Label>
   336→                  <Input
   337→                    id="invite-email"
   338→                    type="email"
   339→                    placeholder="user@example.com"
   340→                    value={inviteEmail}
   341→                    onChange={(e) => setInviteEmail(e.target.value)}
   342→                  />
   343→                </div>
   344→                <div className="space-y-1.5">
   345→                  <Label htmlFor="invite-name">Display name (optional)</Label>
   346→                  <Input
   347→                    id="invite-name"
   348→                    placeholder="Their full name"
   349→                    value={inviteName}
   350→                    onChange={(e) => setInviteName(e.target.value)}
   351→                  />
   352→                </div>
   353→                <div className="space-y-1.5">
   354→                  <Label>Role *</Label>
   355→                  <Select value={inviteRole} onValueChange={setInviteRole}>
   356→                    <SelectTrigger><SelectValue /></SelectTrigger>
   357→                    <SelectContent>
   358→                      <SelectItem value="seeker">Job seeker</SelectItem>
   359→                      <SelectItem value="employer">Employer</SelectItem>
   360→                      <SelectItem value="admin">Administrator</SelectItem>
   361→                    </SelectContent>
   362→                  </Select>
   363→                </div>
   364→                <div className="rounded-md border border-blue-200 bg-blue-50 px-3 py-2 text-[11px] text-blue-700">
   365→                  <Mail className="inline h-3 w-3 mr-1" />
   366→                  In production, an email with a signup link would be sent to <strong>{inviteEmail || "this address"}</strong>.
   367→                </div>
   368→              </div>
   369→              <DialogFooter>
   370→                <Button variant="outline" onClick={() => setInviteOpen(false)}>Cancel</Button>
   371→                <Button className="btn-blue" onClick={sendInvite} disabled={inviting || !inviteEmail.trim()}>
   372→                  {inviting ? (
   373→                    <>
   374→                      <span className="h-3.5 w-3.5 animate-spin rounded-full border-2 border-primary-foreground border-t-transparent" />
   375→                      Sending…
   376→                    </>
   377→                  ) : (
   378→                    <>
   379→                      <Mail className="h-3.5 w-3.5" /> Send invitation
   380→                    </>
   381→                  )}
   382→                </Button>
   383→              </DialogFooter>
   384→            </DialogContent>
   385→          </Dialog>
   386→        }
   387→      />
   388→
   389→      <div className="mt-6 grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   390→        <StatCard label="Total users" value={(data?.total ?? allUsers.length).toLocaleString()} icon={<Users className="h-4 w-4" />} />
   391→        <StatCard label="Active" value={allUsers.filter(u => u.status === "active").length} icon={<UserCheck className="h-4 w-4" />} hint="of filtered" />
   392→        <StatCard label="Pending" value={allUsers.filter(u => u.status === "pending").length} icon={<Clock className="h-4 w-4" />} />
   393→        <StatCard label="Suspended" value={allUsers.filter(u => u.status === "suspended").length} icon={<UserX className="h-4 w-4" />} />
   394→      </div>
   395→
   396→      <Card className="mt-6">
   397→        <CardContent className="p-4">
   398→          <TableToolbar>
   399→            <div className="flex flex-1 flex-col gap-2 sm:flex-row">
   400→              <div className="relative flex-1 max-w-md">
   401→                <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   402→                <Input
   403→                  placeholder="Search by name or email… (debounced)"
   404→                  className="pl-9"
   405→                  value={query}
   406→                  onChange={(e) => setQuery(e.target.value)}
   407→                />
   408→              </div>
   409→              <Select value={roleFilter} onValueChange={(v) => { setRoleFilter(v); setPage(1); }}>
   410→                <SelectTrigger className="w-40"><SelectValue placeholder="Role" /></SelectTrigger>
   411→                <SelectContent>
   412→                  <SelectItem value="all">All roles</SelectItem>
   413→                  <SelectItem value="seeker">Job seekers</SelectItem>
   414→                  <SelectItem value="employer">Employers</SelectItem>
   415→                  <SelectItem value="admin">Admins</SelectItem>
   416→                </SelectContent>
   417→              </Select>
   418→              <Select value={filter} onValueChange={(v) => { setFilter(v); setPage(1); }}>
   419→                <SelectTrigger className="w-40"><SelectValue placeholder="Status" /></SelectTrigger>
   420→                <SelectContent>
   421→                  <SelectItem value="all">All statuses</SelectItem>
   422→                  <SelectItem value="active">Active</SelectItem>
   423→                  <SelectItem value="pending">Pending</SelectItem>
   424→                  <SelectItem value="suspended">Suspended</SelectItem>
   425→                </SelectContent>
   426→              </Select>
   427→            </div>
   428→          </TableToolbar>
   429→
   430→          {loading ? (
   431→            <LoadingState label="Loading users…" rows={4} />
   432→          ) : paged.length === 0 ? (
   433→            <EmptyState
   434→              className="mt-4"
   435→              icon={<Users className="h-5 w-5" />}
   436→              title="No users match your filters"
   437→              description="Try adjusting search or filters."
   438→            />
   439→          ) : (
   440→            <div className="mt-4 overflow-x-auto rounded-lg border border-border">
   441→              <table className="w-full text-sm">
   442→                <thead className="bg-muted/50 text-[11px] uppercase tracking-wide text-muted-foreground">
   443→                  <tr>
   444→                    <th className="px-4 py-2.5 text-left font-medium">User</th>
   445→                    <th className="hidden px-4 py-2.5 text-left font-medium sm:table-cell">Role</th>
   446→                    <th className="hidden px-4 py-2.5 text-left font-medium md:table-cell">Joined</th>
   447→                    <th className="hidden px-4 py-2.5 text-left font-medium lg:table-cell">Last active</th>
   448→                    <th className="px-4 py-2.5 text-left font-medium">Status</th>
   449→                    <th className="px-4 py-2.5 text-right font-medium">Actions</th>
   450→                  </tr>
   451→                </thead>
   452→                <tbody className="divide-y divide-border bg-card">
   453→                  {paged.map((u) => (
   454→                    <tr key={u.id} className="hover:bg-accent/30">
   455→                      <td className="px-4 py-3">
   456→                        <div className="flex items-center gap-2.5">

   459→                            <p className="truncate text-sm font-semibold">{u.name}</p>
   460→                            <p className="truncate text-xs text-muted-foreground">{u.email}</p>
   461→                          </div>
   462→                        </div>
   463→                      </td>
   464→                      <td className="hidden px-4 py-3 sm:table-cell">
   465→                        <Badge variant="outline" className={cn(
   466→                          "capitalize text-[10px]",
   467→                          u.role === "employer" ? "bg-blue-50 text-blue-700 border-blue-200" :

   469→                          "bg-amber-50 text-amber-700 border-amber-200"
   470→                        )}>

   476→                      <td className="px-4 py-3">
   477→                        <Badge variant="outline" className={cn(

   479→                          u.status === "active" ? "bg-emerald-50 text-emerald-700 border-emerald-200" :

   485→                      </td>
   486→                      <td className="px-4 py-3 text-right">
   487→                        <div className="flex justify-end gap-1">
   488→                          <Button size="sm" variant="ghost" onClick={() => setSelected(u)} title="View / edit">
   489→                            <Edit className="h-3.5 w-3.5" />
