   128→function timeAgo(date: string) {
   129→  const d = new Date(date);
   130→  const days = Math.floor((Date.now() - d.getTime()) / (1000 * 60 * 60 * 24));
   131→  if (days === 0) return "today";
   132→  if (days === 1) return "1d ago";
   133→  if (days < 7) return `${days}d ago`;
   134→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   135→  return `${Math.floor(days / 30)}mo ago`;
   136→}
