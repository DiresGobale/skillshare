   226→// ============================================================
   227→// MOBILE NAVBAR — slide-in panel controlled by hamburger toggle
   228→// ============================================================
   229→function MobileNavbar({
   230→  open,
   231→  onClose,
   232→  role,
   233→  view,
   234→  onNavigate,
   235→}: {
   236→  open: boolean;
   237→  onClose: () => void;
   238→  role: Role;
   239→  view: View;
   240→  onNavigate: (v: View) => void;
   241→}) {
   242→  const nav = getNav(role);
   243→  const { user } = useApp();
   244→  const { data: profileData } = useFetch<any>("/api/profile");
   245→  const profilePhoto = profileData?.profile?.profilePhoto;
   246→  useEffect(() => {
   247→    if (open) {
   248→      document.body.style.overflow = "hidden";
   249→    } else {
   250→      document.body.style.overflow = "";
   251→    }
   252→    return () => {
   253→      document.body.style.overflow = "";
   254→    };
   255→  }, [open]);
   256→
   257→  // Close on Escape
   258→  useEffect(() => {
   259→    if (!open) return;
   260→    const handler = (e: KeyboardEvent) => {
   261→      if (e.key === "Escape") onClose();
   262→    };
   263→    window.addEventListener("keydown", handler);
   264→    return () => window.removeEventListener("keydown", handler);
   265→  }, [open, onClose]);
   266→
   267→  return (
   268→    <>
   269→      {/* Backdrop */}
   270→      <div
   271→        className={cn(
   272→          "fixed inset-0 z-40 bg-black/50 backdrop-blur-sm transition-opacity duration-300 lg:hidden",
   273→          open ? "opacity-100" : "pointer-events-none opacity-0"
   274→        )}
   275→        onClick={onClose}
   276→        aria-hidden="true"
   277→      />
   278→
   279→      {/* Slide-in panel */}
   280→      <nav
   281→        id="mobile-navbar"
   282→        aria-label="Mobile navigation"
   283→        className={cn(
   284→          "fixed inset-y-0 left-0 z-50 flex w-72 max-w-[85vw] flex-col border-r border-sidebar-border bg-sidebar shadow-xl transition-transform duration-300 ease-in-out lg:hidden",
   285→          open ? "translate-x-0" : "-translate-x-full"
   286→        )}
   287→      >
   288→        {/* Header with brand + close button */}
   289→        <div className="flex h-16 items-center justify-between border-b border-sidebar-border px-5">
   290→          <BrandMark />
   291→          <button
   292→            onClick={onClose}
   293→            aria-label="Close menu"
   294→            className="inline-flex h-8 w-8 items-center justify-center rounded-md text-muted-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   295→          >
   296→            <X className="h-4 w-4" />
   297→          </button>
   298→        </div>
   299→
   300→        {/* Nav items */}
   301→        <div className="flex-1 overflow-y-auto px-3 py-4">
   302→          <p className="px-3 pb-2 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   303→            {role === "seeker" ? "Job seeker" : role === "employer" ? "Employer" : "Administrator"}
   304→          </p>
   305→          <div className="space-y-0.5">
   306→            {nav.map((item) => {
   307→              const active = view === item.view;
   308→              return (
   309→                <button
   310→                  key={item.view}
   311→                  onClick={() => {
   312→                    onNavigate(item.view);
   313→                    onClose();
   314→                  }}
   315→                  className={cn(
   316→                    "group flex w-full items-center gap-2.5 rounded-md px-3 py-2.5 text-sm font-medium transition-colors",
   317→                    active
   318→                      ? "bg-primary text-primary-foreground shadow-sm"
   319→                      : "text-sidebar-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   320→                  )}
   321→                >
   322→                  <span className={cn(active ? "text-primary-foreground" : "text-muted-foreground group-hover:text-sidebar-accent-foreground")}>
   323→                    {item.icon}
   324→                  </span>
   325→                  <span className="flex-1 text-left">{item.label}</span>
   326→                  {item.badge && (
   327→                    <span
   328→                      className={cn(
   329→                        "inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[10px] font-semibold",
   330→                        active ? "bg-primary-foreground/20 text-primary-foreground" : "bg-primary text-primary-foreground"
   331→                      )}
   332→                    >
   333→                      {item.badge}
   334→                    </span>
   335→                  )}
   336→                </button>
   337→              );
   338→            })}
   339→          </div>
   340→        </div>
   341→
   342→        {/* User card */}
   343→        {user && (
   344→          <div className="border-t border-sidebar-border p-3">
   345→            <div className="flex items-center gap-2.5 rounded-md px-2 py-2">
   346→              <Avatar initials={user.initials ?? user.name.slice(0,2).toUpperCase()} color={user.avatarColor} size="sm" photo={profilePhoto} />
   347→              <div className="min-w-0 flex-1">
   348→                <p className="truncate text-xs font-semibold">{user.name}</p>
   349→                <p className="truncate text-[10px] text-muted-foreground">{user.title ?? user.email}</p>
   350→              </div>
   351→            </div>
   352→          </div>
   353→        )}
   354→      </nav>
   355→    </>
   356→  );
   357→}

   519→function MobileBottomNav({ role, view, onNavigate }: { role: Role; view: View; onNavigate: (v: View) => void }) {
   520→  // Pick the 4 most important views per role for the bottom bar
   521→  const items: { view: View; label: string; icon: React.ReactNode }[] =
   522→    role === "seeker"
   523→      ? [
   524→          { view: "dashboard", label: "Home", icon: <LayoutDashboard className="h-5 w-5" /> },

   527→          { view: "messages", label: "Messages", icon: <MessageSquare className="h-5 w-5" /> },

   530→      ? [

   539→          { view: "jobs", label: "Jobs", icon: <Briefcase className="h-5 w-5" /> },
   540→          { view: "activity", label: "Activity", icon: <Activity className="h-5 w-5" /> },
   541→        ];

   543→  return (

   545→      aria-label="Mobile bottom navigation"
   546→      className="fixed inset-x-0 bottom-0 z-30 flex h-16 items-stretch border-t border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80 lg:hidden"
   547→      style={{ paddingBottom: "env(safe-area-inset-bottom, 0px)" }}

   576→// ============================================================
   577→// MAIN LAYOUT — sidebar + topbar + content
   578→// ============================================================
   579→export function AppShell({ children }: { children: React.ReactNode }) {
   580→  const { role, view, go } = useApp();
   581→  const [mobileOpen, setMobileOpen] = useState(false);
   582→  const [showNavHint, setShowNavHint] = useState(false);
   583→
   584→  // After login, briefly auto-open the mobile sidebar so users on small
   585→  // screens can see that navigation exists. We check a sessionStorage flag
   586→  // set by the login flow, then clear it. Only triggers on mobile (lg down).
   587→  useEffect(() => {
   588→    try {
   589→      if (sessionStorage.getItem("skillmatch.justLoggedIn") === "1") {
   590→        sessionStorage.removeItem("skillmatch.justLoggedIn");
   591→        // Only auto-open on mobile (window width < 1024 = lg breakpoint)
   592→        if (typeof window !== "undefined" && window.innerWidth < 1024) {
   593→          // Small delay so the dashboard content renders first
   594→          const t1 = setTimeout(() => setMobileOpen(true), 600);
   595→          // Auto-close after 4 seconds if the user hasn't interacted
   596→          const t2 = setTimeout(() => setMobileOpen(false), 4600);
   597→          // Show a pulsing hint on the hamburger after the sidebar closes
   598→          const t3 = setTimeout(() => setShowNavHint(true), 4800);
   599→          const t4 = setTimeout(() => setShowNavHint(false), 10000);
   600→          return () => {
   601→            clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4);
   602→          };
   603→        }
   604→      }
   605→    } catch { /* ignore */ }
   606→  }, []);
   607→
   608→  return (
   609→    <div className="flex min-h-screen bg-background">
   610→      {/* Desktop sidebar */}
   611→      <aside className="hidden w-64 shrink-0 border-r border-sidebar-border bg-sidebar lg:flex lg:flex-col">
   612→        <SidebarNav role={role} view={view} onNavigate={(v) => go(v)} />
   613→      </aside>
   614→
   615→      {/* Mobile navbar (toggle-controlled, slide-in) */}
   616→      <MobileNavbar
   617→        open={mobileOpen}
   618→        onClose={() => setMobileOpen(false)}
   619→        role={role}
   620→        view={view}
   621→        onNavigate={(v) => go(v)}
   622→      />
   623→
   624→      {/* Main */}
   625→      <div className="flex min-w-0 flex-1 flex-col">
   626→        <Topbar
   627→          onToggleMobile={() => { setMobileOpen((v) => !v); setShowNavHint(false); }}
   628→          mobileOpen={mobileOpen}
   629→          showNavHint={showNavHint}
   630→        />
   631→        {/* pb-20 on mobile to account for the bottom nav bar; lg:pb-0 to remove it on desktop */}
   632→        <main className="flex-1 pb-20 lg:pb-0">{children}</main>
   633→      </div>
   634→
   635→      {/* Mobile bottom navigation — persistent quick access to key pages */}
   636→      <MobileBottomNav role={role} view={view} onNavigate={(v) => go(v)} />
   637→    </div>
   638→  );
   639→}
