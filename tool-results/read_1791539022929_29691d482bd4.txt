   229→   229→function MobileNavbar({
   230→   230→  open,
   231→   231→  onClose,
   232→   232→  role,
   233→   233→  view,
   234→   234→  onNavigate,
   235→   235→}: {
   236→   236→  open: boolean;
   237→   237→  onClose: () => void;
   238→   238→  role: Role;
   239→   239→  view: View;
   240→   240→  onNavigate: (v: View) => void;
   241→   241→}) {
   242→   242→  const nav = getNav(role);
   243→   243→  const { user } = useApp();
   244→   244→  const { data: profileData } = useFetch<any>("/api/profile");
   245→   245→  const profilePhoto = profileData?.profile?.profilePhoto;
   246→   246→  useEffect(() => {
   247→   247→    if (open) {
   248→   248→      document.body.style.overflow = "hidden";
   249→   249→    } else {
   250→   250→      document.body.style.overflow = "";
   251→   251→    }
   252→   252→    return () => {
   253→   253→      document.body.style.overflow = "";
   254→   254→    };
   255→   255→  }, [open]);
   256→   256→
   257→   257→  // Close on Escape
   258→   258→  useEffect(() => {
   259→   259→    if (!open) return;
   260→   260→    const handler = (e: KeyboardEvent) => {
   261→   261→      if (e.key === "Escape") onClose();
   262→   262→    };
   263→   263→    window.addEventListener("keydown", handler);
   264→   264→    return () => window.removeEventListener("keydown", handler);
   265→   265→  }, [open, onClose]);
   266→   266→
   267→   267→  return (
   268→   268→    <>
   269→   269→      {/* Backdrop */}
   270→   270→      <div
   271→   271→        className={cn(
   272→   272→          "fixed inset-0 z-40 bg-black/50 backdrop-blur-sm transition-opacity duration-300 lg:hidden",
   273→   273→          open ? "opacity-100" : "pointer-events-none opacity-0"
   274→   274→        )}
   275→   275→        onClick={onClose}
   276→   276→        aria-hidden="true"
   277→   277→      />
   278→   278→
   279→   279→      {/* Slide-in panel */}
   280→   280→      <nav
   281→   281→        id="mobile-navbar"
   282→   282→        aria-label="Mobile navigation"
   283→   283→        className={cn(
   284→   284→          "fixed inset-y-0 left-0 z-50 flex w-72 max-w-[85vw] flex-col border-r border-sidebar-border bg-sidebar shadow-xl transition-transform duration-300 ease-in-out lg:hidden",
   285→   285→          open ? "translate-x-0" : "-translate-x-full"
   286→   286→        )}
   287→   287→      >
   288→   288→        {/* Header with brand + close button */}
   289→   289→        <div className="flex h-16 items-center justify-between border-b border-sidebar-border px-5">
   290→   290→          <BrandMark />
   291→   291→          <button
   292→   292→            onClick={onClose}
   293→   293→            aria-label="Close menu"
   294→   294→            className="inline-flex h-8 w-8 items-center justify-center rounded-md text-muted-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
   295→   295→          >
   296→   296→            <X className="h-4 w-4" />
   297→   297→          </button>
   298→   298→        </div>
   299→   299→
   300→   300→        {/* Nav items */}
   301→   301→        <div className="flex-1 overflow-y-auto px-3 py-4">
   302→   302→          <p className="px-3 pb-2 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   303→   303→            {role === "seeker" ? "Job seeker" : role === "employer" ? "Employer" : "Administrator"}
   304→   304→          </p>
   305→   305→          <div className="space-y-0.5">
   306→   306→            {nav.map((item) => {
   307→   307→              const active = view === item.view;
   308→   308→              return (
   309→   309→                <button
   310→   310→                  key={item.view}

   317→   317→                    active
   318→   318→                      ? "bg-primary text-primary-foreground shadow-sm"

   579→   579→export function AppShell({ children }: { children: React.ReactNode }) {
   580→   580→  const { role, view, go } = useApp();
   581→   581→  const [mobileOpen, setMobileOpen] = useState(false);
   582→   582→  const [showNavHint, setShowNavHint] = useState(false);
   583→   583→
   584→   584→  // After login, briefly auto-open the mobile sidebar so users on small
   585→   585→  // screens can see that navigation exists. We check a sessionStorage flag
   586→   586→  // set by the login flow, then clear it. Only triggers on mobile (lg down).
   587→   587→  useEffect(() => {
   588→   588→    try {
   589→   589→      if (sessionStorage.getItem("skillmatch.justLoggedIn") === "1") {
   590→   590→        sessionStorage.removeItem("skillmatch.justLoggedIn");
   591→   591→        // Only auto-open on mobile (window width < 1024 = lg breakpoint)
   592→   592→        if (typeof window !== "undefined" && window.innerWidth < 1024) {
   593→   593→          // Small delay so the dashboard content renders first
   594→   594→          const t1 = setTimeout(() => setMobileOpen(true), 600);
   595→   595→          // Auto-close after 4 seconds if the user hasn't interacted
   596→   596→          const t2 = setTimeout(() => setMobileOpen(false), 4600);
   597→   597→          // Show a pulsing hint on the hamburger after the sidebar closes
   598→   598→          const t3 = setTimeout(() => setShowNavHint(true), 4800);
   599→   599→          const t4 = setTimeout(() => setShowNavHint(false), 10000);
   600→   600→          return () => {
   601→   601→            clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4);
   602→   602→          };
   603→   603→        }
   604→   604→      }
   605→   605→    } catch { /* ignore */ }
   606→   606→  }, []);
   607→   607→
   608→   608→  return (
   609→   609→    <div className="flex min-h-screen bg-background">
   610→   610→      {/* Desktop sidebar */}
   611→   611→      <aside className="hidden w-64 shrink-0 border-r border-sidebar-border bg-sidebar lg:flex lg:flex-col">
   612→   612→        <SidebarNav role={role} view={view} onNavigate={(v) => go(v)} />
   613→   613→      </aside>
   614→   614→
   615→   615→      {/* Mobile navbar (toggle-controlled, slide-in) */}
   616→   616→      <MobileNavbar
   617→   617→        open={mobileOpen}
   618→   618→        onClose={() => setMobileOpen(false)}
   619→   619→        role={role}
   620→   620→        view={view}
   621→   621→        onNavigate={(v) => go(v)}
   622→   622→      />
   623→   623→
   624→   624→      {/* Main */}
   625→   625→      <div className="flex min-w-0 flex-1 flex-col">
   626→   626→        <Topbar
   627→   627→          onToggleMobile={() => { setMobileOpen((v) => !v); setShowNavHint(false); }}
   628→   628→          mobileOpen={mobileOpen}
   629→   629→          showNavHint={showNavHint}
   630→   630→        />
   631→   631→        {/* pb-20 on mobile to account for the bottom nav bar; lg:pb-0 to remove it on desktop */}
   632→   632→        <main className="flex-1 pb-20 lg:pb-0">{children}</main>
   633→   633→      </div>
   634→   634→
   635→   635→      {/* Mobile bottom navigation — persistent quick access to key pages */}
   636→   636→      <MobileBottomNav role={role} view={view} onNavigate={(v) => go(v)} />
   637→   637→    </div>
   638→   638→  );
   639→   639→}
